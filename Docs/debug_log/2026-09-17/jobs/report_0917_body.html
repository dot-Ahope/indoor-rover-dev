def split_x(c):
    o = c[np.argsort(c[:, 0])]; parts, cur2 = [], [o[0]]
    for q in o[1:]:
        if q[0] - cur2[-1][0] > 0.10:          # 10 cm 넘는 x 간격에서 나눔
            parts.append(np.array(cur2)); cur2 = [q]
        else:
            cur2.append(q)
    parts.append(np.array(cur2))
    best = max(parts, key=len)                  # 점이 가장 많은 조각만
    return best, len(c) - len(best)
